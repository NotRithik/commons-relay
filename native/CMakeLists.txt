# Raw native installations need the same manifest as extracted LGX packages.
# Do not depend on metadata left by an older deployment on the build machine.
if(APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
 set(RELAY_NATIVE_VARIANT "darwin-arm64")
elseif(APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
 set(RELAY_NATIVE_VARIANT "darwin-amd64")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
 set(RELAY_NATIVE_VARIANT "linux-amd64")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
 set(RELAY_NATIVE_VARIANT "linux-arm64")
else()
 message(FATAL_ERROR "No Logos package variant for ${CMAKE_SYSTEM_NAME}/${CMAKE_SYSTEM_PROCESSOR}")
endif()
function(relay_install_manifest module metadata)
 file(READ "${metadata}" manifest)
 string(JSON manifest SET "${manifest}" manifestVersion "\"0.3.0\"")
 string(JSON manifest SET "${manifest}" main "{}")
 string(JSON manifest SET "${manifest}" main "${RELAY_NATIVE_VARIANT}" "\"${module}_plugin${CMAKE_SHARED_LIBRARY_SUFFIX}\"")
 file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/manifests/${module}")
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/manifests/${module}/manifest.json" "${manifest}\n")
 install(FILES "${CMAKE_CURRENT_BINARY_DIR}/manifests/${module}/manifest.json" DESTINATION "${module}")
endfunction()
relay_install_manifest(commons_relay_module "${CMAKE_CURRENT_SOURCE_DIR}/core/metadata.json")
relay_install_manifest(commons_relay_wallet "${CMAKE_CURRENT_SOURCE_DIR}/wallet/metadata.json")
relay_install_manifest(commons_relay_owner_ui "${CMAKE_CURRENT_SOURCE_DIR}/ui/metadata.json")
