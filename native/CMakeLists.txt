cmake_minimum_required(VERSION 3.20)
project(CommonsCommonsRelayNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.9 REQUIRED COMPONENTS Core RemoteObjects Qml Test)
set(LOGOS_DEPS_DIR "" CACHE PATH "Pinned Logos protocol and plugin checkouts")
if(NOT IS_DIRECTORY "${LOGOS_DEPS_DIR}/logos-plugin-qt")
 message(FATAL_ERROR "Set LOGOS_DEPS_DIR to the pinned Logos dependencies")
endif()
find_path(NLOHMANN_INCLUDE_DIR nlohmann/json.hpp PATHS /opt/homebrew/include REQUIRED)
add_library(commons_relay_module_plugin SHARED core/commons_relay_module.h core/commons_relay_module.cpp)
set_target_properties(commons_relay_module_plugin PROPERTIES PREFIX "" INSTALL_RPATH "@executable_path/../Frameworks")
target_include_directories(commons_relay_module_plugin PRIVATE core "${NLOHMANN_INCLUDE_DIR}" "${LOGOS_DEPS_DIR}/logos-plugin-qt/core" "${LOGOS_DEPS_DIR}/logos-plugin-qt/cpp" "${LOGOS_DEPS_DIR}/logos-protocol/cpp")
target_link_libraries(commons_relay_module_plugin PRIVATE Qt6::Core Qt6::RemoteObjects Qt6::Qml)
if(APPLE)
 target_link_options(commons_relay_module_plugin PRIVATE -undefined dynamic_lookup)
else()
 target_link_libraries(commons_relay_module_plugin PRIVATE dl)
endif()
install(TARGETS commons_relay_module_plugin LIBRARY DESTINATION commons_relay_module)
install(FILES core/metadata.json DESTINATION commons_relay_module)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../commons_relay_worker.py" DESTINATION commons_relay_module)
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../commons_relay" DESTINATION commons_relay_module FILES_MATCHING PATTERN "*.py")
find_package(Qt6 6.9 REQUIRED COMPONENTS Gui Quick)
set(LOGOS_VIEW_GENERATOR "" CACHE FILEPATH "Official logos-view-generator")
if(NOT EXISTS "${LOGOS_VIEW_GENERATOR}")
 message(FATAL_ERROR "Set LOGOS_VIEW_GENERATOR")
endif()
set(MOD "${CMAKE_CURRENT_SOURCE_DIR}/ui")
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN}")
configure_file("${MOD}/metadata.json" "${GEN}/metadata.json" COPYONLY)
execute_process(COMMAND "${LOGOS_VIEW_GENERATOR}" --backend ui --metadata "${MOD}/metadata.json" --rep "${MOD}/src/commons_relay_owner_ui.rep" --backend-class CommonsRelayUiBackend --backend-header commons_relay_ui_backend.h --output-dir "${GEN}" COMMAND_ERROR_IS_FATAL ANY)
set(LOGOS_REP_CLASS CommonsRelayOwnerUi)
set(LOGOS_REP_BASE commons_relay_owner_ui)
set(LOGOS_FACTORY_CLASS CommonsRelayOwnerUiReplicaFactory)
set(LOGOS_QML_URI Logos.CommonsRelay)
set(LOGOS_QML_TYPE_NAME CommonsRelay)
foreach(name LogosViewPluginBase.h LogosViewPluginBase.cpp LogosViewReplicaFactory.h LogosViewReplicaFactory.cpp)
 configure_file("${LOGOS_DEPS_DIR}/logos-view-module/cmake/${name}.in" "${GEN}/${name}" @ONLY)
endforeach()
# Typed dependency context owned by the host; no second LogosAPI is created.
file(WRITE "${GEN}/logos_sdk.h" "#pragma once\nclass LogosAPI;\nstruct LogosModules { LogosAPI* api; explicit LogosModules(LogosAPI* host):api(host){} };\n")
add_library(commons_relay_owner_ui_plugin SHARED "${MOD}/src/commons_relay_ui_backend.h" "${MOD}/src/commons_relay_ui_backend.cpp" "${GEN}/commons_relay_owner_ui_ui_glue.h" "${GEN}/commons_relay_owner_ui_ui_glue.cpp" "${GEN}/LogosViewPluginBase.h" "${GEN}/LogosViewPluginBase.cpp")
set_target_properties(commons_relay_owner_ui_plugin PROPERTIES PREFIX "" INSTALL_RPATH "@executable_path/../Frameworks")
target_include_directories(commons_relay_owner_ui_plugin PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${MOD}/src" "${GEN}" "${NLOHMANN_INCLUDE_DIR}" "${LOGOS_DEPS_DIR}/logos-view-module/cpp" "${LOGOS_DEPS_DIR}/logos-plugin-qt/core" "${LOGOS_DEPS_DIR}/logos-plugin-qt/cpp" "${LOGOS_DEPS_DIR}/logos-protocol/cpp")
target_link_libraries(commons_relay_owner_ui_plugin PRIVATE Qt6::Core Qt6::RemoteObjects Qt6::Qml)
qt6_add_repc_sources(commons_relay_owner_ui_plugin "${MOD}/src/commons_relay_owner_ui.rep")
if(APPLE)
 target_link_options(commons_relay_owner_ui_plugin PRIVATE -undefined dynamic_lookup)
endif()
add_library(commons_relay_owner_ui_replica_factory SHARED "${GEN}/LogosViewReplicaFactory.h" "${GEN}/LogosViewReplicaFactory.cpp")
set_target_properties(commons_relay_owner_ui_replica_factory PROPERTIES PREFIX "")
target_include_directories(commons_relay_owner_ui_replica_factory PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${GEN}")
target_link_libraries(commons_relay_owner_ui_replica_factory PRIVATE Qt6::Core Qt6::RemoteObjects Qt6::Qml)
qt6_add_repc_replicas(commons_relay_owner_ui_replica_factory "${MOD}/src/commons_relay_owner_ui.rep")
install(TARGETS commons_relay_owner_ui_plugin commons_relay_owner_ui_replica_factory LIBRARY DESTINATION commons_relay_owner_ui)
install(FILES "${MOD}/metadata.json" DESTINATION commons_relay_owner_ui)
install(DIRECTORY "${MOD}/src/qml" "${MOD}/icons" DESTINATION commons_relay_owner_ui)
